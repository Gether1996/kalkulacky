# models.py

class UserPreference(models.Model):
    """User notification and preference settings"""
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    email_notifications = models.BooleanField(default=True)
    sms_notifications = models.BooleanField(default=False)
    push_notifications = models.BooleanField(default=True)
    
    notification_frequency = models.CharField(
        max_length=20,
        choices=[
            ('realtime', 'V reálnom čase'),
            ('daily', 'Denne'),
            ('weekly', 'Týždenne'),
            ('monthly', 'Mesačne')
        ],
        default='realtime'
    )
    
    timezone = models.CharField(max_length=50, default='Europe/Bratislava')
    language = models.CharField(max_length=10, default='sk')
    
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)


class SavedCalculation(models.Model):
    """Saved calculator results with tracking"""
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    
    calculator_type = models.CharField(
        max_length=50,
        choices=[
            ('pregnancy', 'Tehotenstvo'),
            ('mortgage', 'Hypotéka'),
            ('loan', 'Úver'),
            ('vacation', 'Dovolenka'),
            ('pension', 'Dôchodok'),
            ('freelancer_tax', 'SZČO dane'),
            ('bmi', 'BMI'),
            ('energy', 'Energia'),
            ('sick_leave', 'Nemocenská'),
            # nové
            ('parental_benefit', 'Rodičovský príspevok'),
            ('child_support', 'Výživné'),
            ('advances', 'Preddavky'),
        ]
    )
    
    name = models.CharField(
        max_length=100,
        help_text="User-defined name (e.g., 'Moja hypotéka VUB')"
    )
    
    params = models.JSONField(
        help_text="Calculator input parameters"
    )
    
    result = models.JSONField(
        null=True,
        blank=True,
        help_text="Cached calculation result"
    )
    
    is_tracking = models.BooleanField(
        default=False,
        help_text="Whether user wants notifications for this calculation"
    )
    
    is_favorite = models.BooleanField(default=False)
    
    created_at = models.DateTimeField(auto_now_add=True)
    last_accessed = models.DateTimeField(auto_now=True)
    
    class Meta:
        ordering = ['-last_accessed']
        indexes = [
            models.Index(fields=['user', 'calculator_type']),
            models.Index(fields=['user', 'is_tracking']),
        ]


class ScheduledNotification(models.Model):
    """Scheduled notifications for tracked calculations"""
    calculation = models.ForeignKey(
        SavedCalculation,
        on_delete=models.CASCADE,
        related_name='notifications'
    )
    
    notification_type = models.CharField(
        max_length=50,
        choices=[
            # Pregnancy
            ('pregnancy_week', 'Týždenná aktualizácia tehotenstva'),
            ('pregnancy_milestone', 'Míľnik tehotenstva'),
            ('trimester_change', 'Zmena trimestra'),
            
            # Vacation
            ('vacation_expiry', 'Dovolenka prepadne'),
            ('vacation_quarterly', 'Štvrťročná kontrola dovolenky'),
            ('birthday_33', 'Prírastok dovolenky (33 rokov)'),
            
            # Mortgage/Loan
            ('payment_due', 'Splátka splatná'),
            ('rate_change', 'Zmena úrokovej sadzby'),
            ('milestone_paid', 'Míľnik splatenia'),
            
            # Tax
            ('tax_deadline', 'Daňový deadline'),
            ('advance_due', 'Preddavok splatný'),
            ('threshold_warning', 'Blížite sa k daňovému prahu'),
            
            # Pension
            ('pension_annual', 'Ročná kontrola dôchodku'),
            ('pension_milestone', 'Míľnik pred dôchodkom'),
            
            # BMI/Fitness
            ('weight_checkin', 'Týždenné váženie'),
            ('goal_milestone', 'Míľnik cieľa'),
            
            # Energy
            ('energy_monthly', 'Mesačná spotreba energie'),
            ('rate_increase', 'Zdraženie energie'),
            
            # Sick leave
            ('sick_leave_daily', 'Denný update nemocenskej'),
            ('sick_leave_end', 'Koniec nemocenskej'),
            
            # Parental benefit
            ('parental_expiry', 'Koniec rodičovského príspevku'),
            ('work_return', 'Návrat do práce'),
        ]
    )
    
    scheduled_date = models.DateField()
    scheduled_time = models.TimeField(default='09:00:00')
    
    message_title = models.CharField(max_length=200)
    message_body = models.TextField()
    
    sent = models.BooleanField(default=False)
    sent_at = models.DateTimeField(null=True, blank=True)
    
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        ordering = ['scheduled_date', 'scheduled_time']
        indexes = [
            models.Index(fields=['scheduled_date', 'sent']),
            models.Index(fields=['calculation', 'notification_type']),
        ]


class NotificationHistory(models.Model):
    """Log of all sent notifications"""
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    notification = models.ForeignKey(
        ScheduledNotification,
        on_delete=models.SET_NULL,
        null=True,
        blank=True
    )
    
    notification_type = models.CharField(max_length=50)

from django.core.management.base import BaseCommand
from django.utils import timezone
from calculators.models import ScheduledNotification, NotificationHistory
from calculators.services.notification_service import NotificationService


class Command(BaseCommand):
    help = 'Send scheduled notifications'
    
    def handle(self, *args, **options):
        now = timezone.now()
        today = now.date()
        current_time = now.time()
        
        # Get notifications scheduled for today that haven't been sent
        notifications = ScheduledNotification.objects.filter(
            scheduled_date=today,
            scheduled_time__lte=current_time,
            sent=False
        ).select_related('calculation', 'calculation__user')
        
        sent_count = 0
        failed_count = 0
        
        for notification in notifications:
            user = notification.calculation.user
            preferences = user.userpreference
            
            success = False
            
            # Send via preferred channels
            if preferences.email_notifications:
                success = NotificationService.send_email(
                    user,
                    notification.message_title,
                    notification.message_body
                )
                
                if success:
                    NotificationHistory.objects.create(
                        user=user,
                        notification=notification,
                        notification_type=notification.notification_type,
                        message_title=notification.message_title,
                        message_body=notification.message_body,
                        channel='email',
                        status='sent'
                    )
            
            if preferences.sms_notifications:
                NotificationService.send_sms(user, notification.message_body)
            
            if preferences.push_notifications:
                NotificationService.send_push(
                    user,
                    notification.message_title,
                    notification.message_body
                )
            
            # Mark as sent
            notification.sent = True
            notification.sent_at = now
            notification.save()
            
            sent_count += 1 if success else 0
            failed_count += 0 if success else 1
        
        self.stdout.write(
            self.style.SUCCESS(
                f'Sent {sent_count} notifications, {failed_count} failed'
            )
        )

// services/local-storage.service.ts

export interface SavedCalculation {
  id: string;
  calculatorType: string;
  name: string;
  params: any;
  result: any;
  timestamp: number;
  isFavorite: boolean;
}

export class LocalStorageService {
  private static readonly STORAGE_KEY = 'kalkulacky_history';
  private static readonly MAX_ITEMS = 10;
  
  static saveCalculation(
    calculatorType: string,
    name: string,
    params: any,
    result: any
  ): void {
    const calculations = this.getCalculations();
    
    const newCalculation: SavedCalculation = {
      id: this.generateId(),
      calculatorType,
      name,
      params,
      result,
      timestamp: Date.now(),
      isFavorite: false
    };
    
    calculations.unshift(newCalculation);
    
    // Keep only last 10 items (or favorites)
    const filtered = calculations
      .filter(c => c.isFavorite)
      .concat(
        calculations
          .filter(c => !c.isFavorite)
          .slice(0, this.MAX_ITEMS)
      );
    
    localStorage.setItem(this.STORAGE_KEY, JSON.stringify(filtered));
  }
  
  static getCalculations(calculatorType?: string): SavedCalculation[] {
    const data = localStorage.getItem(this.STORAGE_KEY);
    if (!data) return [];
    
    const calculations = JSON.parse(data) as SavedCalculation[];
    
    if (calculatorType) {
      return calculations.filter(c => c.calculatorType === calculatorType);
    }
    
    return calculations;
  }
  
  static deleteCalculation(id: string): void {
    const calculations = this.getCalculations();
    const filtered = calculations.filter(c => c.id !== id);
    localStorage.setItem(this.STORAGE_KEY, JSON.stringify(filtered));
  }
  
  static toggleFavorite(id: string): void {
    const calculations = this.getCalculations();
    const calculation = calculations.find(c => c.id === id);
    if (calculation) {
      calculation.isFavorite = !calculation.isFavorite;
      localStorage.setItem(this.STORAGE_KEY, JSON.stringify(calculations));
    }
  }
  
  private static generateId(): string {
    return `calc_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }
}
